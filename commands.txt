  --index-name        quora_di_bench \

  --token             12345678 \
  --base-url          http://51.89.231.115:8080/api/v1 \
  --space-type        cosine \
  --precision         float32 \
  --dimension         384 \
  --query-concurrency 4 \
  --query-async-concurrency 10 \
  --top-k             10 \
  --validation-env    validation-env \
  --output-file       quora_di_results.json
